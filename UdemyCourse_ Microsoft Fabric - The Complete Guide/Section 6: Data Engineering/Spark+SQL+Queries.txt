result_df = spark.sql("""
SELECT 
    p.Product_Name,
    COUNT(s.Transaction_ID) as Total_Sales
FROM 
    LakehouseTraining.sales s
JOIN 
    LakehouseTraining.products p ON s.Product_ID = p.Product_ID
GROUP BY 
    p.Product_Name
HAVING COUNT(s.Transaction_ID) >10
""")
